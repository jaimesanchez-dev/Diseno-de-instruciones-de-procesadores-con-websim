# Extension: new instructions

lx reg1 U32 {
  						co =  100110,
  						nwords = 2, #64_bits
  						reg1 = reg(25,21), #Selc
              U32 = imm(63,32), #valor_de_64
             {
               	(T2, C0), #pc->MAR
              	(TA, R, BW=11, M1=1, C1=1, M2, C2), #Mar->Memory->MBR, pc + 4
                (T1, LC=1, MR=0, SELC=10101, A0=1, B=1, C=0) #Guardar el valor de MBR en el registro
            	}
}




call val {
		co = 100001,
  	       nwords = 2, #64_bits
  	       val = imm(63,32), #valor_de_64 
  {
            (T2, C0), #pc->MAR
            (TA, R, BW=11, M1=1, C1=1, M2, C2), #Mar->Memory->MBR, pc + 4
            (T1, C4), #guardamos el numero en RT1
            
            # Sp <- Sp-4
	(MR=1, SelA=00010, MA=0, MB=10, SelCop=1011, MC, T6, SelC=00010, LC=1, C0, SELP=11, M7, C7) #SeLA tiene ese numero porque la pila esta en  el registro 2 (Tambien guardamos el Sp-4 en C0)
    
            #MEM[SP] <- PC
            (T2, C1), # Pc <-MBR
            (Ta, BW=11, Td, W,T4, M2=0, C2, A0=1, B=1, C=0), # Escribir en memoria (Tenemos Sp en MAR), PC <- RT1
    				
  }				

}

return{
         co = 100010,
  {
        (SelA= 00010, MR, T9, C0), #Devuelve a PC el valor que tenia, que lo tenia la pila guardado
        (Ta, R, BW=11, M1, C1, SelA=00010, MA=0,MB=10, MR, SelCop=1010, MC=1, T6, SelC=00010, LC), #Leemos la dirección de Sp en memoria y le sumamos 4 a la pila
        (T1, M2=0, C2, A0=1, B=1, C=0), #Devolvemos el valor a PC
	}
}

stop{
	co = 100011,
  {
         (SelA=00000, T9, SelC= 00010, MR, LC, M2=0, C2), #Guardamos en la pila el valor "0"(Utilizando el registro 0), Guardamos en el PC el valor "0"(Utilizando el registro 0)
  }
}


vfill reg1 (reg2) val{
     co=100100,
     nwords=1,
     reg1=reg(25,21),
     reg2=reg(20,16),
     val = imm(7,0), #valor_de_8
  
  {
    			(T8, C5) #Guardamos RT2 <- SR
          (SelA=00000, MR=1, T9, C4) #Dejamos vacio RT1
          (SE=1, OFFSET=0, SIZE=01000,T3, M1=0, C1, MB=01, MA=1, SelCop=1010, MC, C6), #Asignamos a MBR el valor U8, y guardamos SR en RT3 (RT3 -< RT2)
          (MR=0,SelB=10000, T10, C5, C0), #RT2 <- reg2, #MAR <- reg2
          
  inicio:	(Td, Ta, BW=00, W, MA=1, SelB=10101, MB=00, SelCop=1011, MC, SelP=11, M7=1,C7), #Guardamos el valor (U8) en la memoria, Restamos R1 e "i", para saber si es 0 y acabar el bucle, Realizamos el salto a fin en caso de que la resta sea 0.
    			(A0=0, B=0, C=110, MADDR=fin, MB=01, MA=1, SelCop=1010, MC, T6, C0) #Realizamos el salto a fin en caso de que la resta sea 0. #RT2 es reg2 (Le sumamos a Reg2 i y lo guardamos en MAR) (I esta en Rt1) Tambien metemos addr en MAR
    
            
  				(MA=1, MB=11, SelCop=1010, MC, T6, C4, A0=0, B=1, C=0, MADDR=inicio), #Actualizamos i y lo dejamos en RT1, Hacemos el salto al inicio.
    
  fin: (SelA=10101, SelB=10000, MA=0, MB=00, SelCop=1010, MR=0, MC, T6, SelC=10000, LC), #Rr2   ←BR[RR2] + BR[RR1]  (RR2 estaba en RT2)
			 (ExCode=0000, T11, SelC=10101, LC), #R1 ahora vale 0(Utilizamos EXCode para obtener un "0") 
    	 (T7, M7=0, C7, A0=1, B=1, C=0)  #devolvemos el valor de SR a SR (Que estaba en RT3)
  }
}

  
  
vadd reg1, (reg2), reg3{
  	co=100101
		nwords=1,
    reg1=reg(25,21),
    reg2=reg(20,16),
    reg3=reg(15,11),
  {
      		(T8, C5) #Guardamos RT2 <- SR
          (SelA=00000, MR=1, T9, C4, MA=0, MB=01, SelCop=1010, MC, C6) #Dejamos vacio RT1 y sumamos 0+SR=SR, para dejarlo en RT3
          
          
          
  inicio2:(MR=0,SelB=10000, MB=00, MA=1, SelCop=1010, MC, T6, C0), #SelB es reg2 (Le sumamos a Reg2 i y lo guardamos en MAR) (I esta en Rt1)
  				(Ta, R, BW=00, M1, C1, SelB=10101, MR=0, MA=1, MB=00, SelCop=1011, MC, SelP=11, M7=1, C7,), #buscamos la addr y gaurdamos el valor en MBR.
  				 #Restamos R1 e "i", para saber si es 0 y acabar el bucle
  				
  				(A0=0, B=0, C=110, MADDR=fin2, T1, C5), #Salto o seguir y #Guardamos el valor de MBR en RT2
      
  
  				(SelA=01011, MR=0, MA=0, MB=01, SelCop=1010, MC, T6, M1=0, C1) #Sumamos el valor + reg3, y lo guardamos en MBR
  								  				
    
  				(Ta, W, Td, BW=00, MA=1, MB=11, SelCop=1010, MC, T6, C4, A0=0, B=1, C=0, MADDR=inicio2), #Actualizamos i, y lo dejamos en RT1 y Guardamos el valor de MBR en la memoria (addr anterior). Realizamos el salto al inicio 
  
  fin2: (SelA=10101, SelB=10000, MA=0, MB=00, SelCop=1010, MR=0, MC, T6, SelC=10000, LC), #Rr2   ←BR[RR2] + BR[RR1]
				(ExCode=0000, T11, SelC=10101, LC), #R1 ahora vale 0(Utilizamos EXCode para obtener un "0") 
  			(T7, M7=0, C7, A0=1, B=1, C=0)  #devolvemos el valor de SR a SR (Que estaba en RT3)
  }
}
